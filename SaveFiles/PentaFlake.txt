iterations: 6 /* choose iteration count */
t5: τ / 5; t10: τ / 10; t10m: τ + t10; scale: φ+; s3 : scale^3 /* constants */
m(a) : (a + t10m) % t5 - t10 /* helper function cycling the angle towards children */
InsidePentagon(z, s, f) : z@ <= scos(t10) / cos(t10 - f@) /* is z inside a pentagon of size s? */
Flake(z, s: 1, n: iterations, /* mandatory and optional inputs */
 l: s / scale, /* child size */
 a: arg(z), /* angle of z from center */
 f: m(a)) /* which outer child we fell onto? */
 : InsidePentagon(z, s, f) = 0 ? _ /* outside the pentagon, or within the voids between them? */
 : n < 0 ? 0 /* bottom of the recursion, initiate the counter */
 : InsidePentagon(-z, l, m(a + π)) ? Flake(-z, l, n-)+ /* central pentagon? it is flipped */ 
 : Flake(z - (zexp(-if))||s / φ, l, n-) /* inside the peripheral 5, translate back to center and recurse */
