/* Newton Fractal */
EvalPoly(c, z, n: c#-) : sum("k", 0, n, "c[k]z^k")
DiffPoly(c, n:c#-) : 1 > n ? 0 : vec("k", 1, n, "kc[k]")

Iterations: 20 /* choose the number of iterations */
Poly : (-1, 0, 0, 1) /* choose the polynomial: z^3 - 1 */

DPoly : DiffPoly(Poly) /* take a derivative */

DrawNewton(v) : hsv2rgb(arg(v) / τ, 1, 1)
Newton(z, n:Iterations, f:EvalPoly(Poly, z), d:EvalPoly(DPoly, z), q: z - f/d) : n < 0 ? z : Newton(q, n-)
